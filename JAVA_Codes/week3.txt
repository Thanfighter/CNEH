import java.util.Scanner;

public class Main {

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter number of routers: ");
        int n = sc.nextInt();

        int[][] cost = new int[n][n];
        int[][] dist = new int[n][n];
        int[][] next = new int[n][n];

        System.out.println("Enter cost matrix (999 for no connection):");

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {

                cost[i][j] = sc.nextInt();

                dist[i][j] = cost[i][j];

                if (i == j) {
                    next[i][j] = i;
                }
                else if (cost[i][j] != 999) {
                    next[i][j] = j;
                }
                else {
                    next[i][j] = -1;
                }
            }
        }

        boolean changed = true;

        while (changed) {

            changed = false;

            for (int i = 0; i < n; i++) {

                for (int j = 0; j < n; j++) {

                    for (int k = 0; k < n; k++) {

                        if (dist[i][k] != 999 &&
                            dist[k][j] != 999) {

                            int newDist = dist[i][k] + dist[k][j];

                            if (newDist < dist[i][j]) {

                                dist[i][j] = newDist;

                                next[i][j] = next[i][k];

                                changed = true;
                            }
                        }
                    }
                }
            }
        }

        System.out.println("\nFinal Routing Tables:");

        for (int i = 0; i < n; i++) {

            System.out.println("\nRouter " + (i + 1));

            System.out.println("Destination\tDistance\tNext Hop");

            for (int j = 0; j < n; j++) {

                System.out.print((j + 1) + "\t\t");

                if (dist[i][j] == 999) {
                    System.out.println("INF\t\t-");
                }
                else {
                    System.out.println(dist[i][j] + "\t\t" +
                        (next[i][j] + 1));
                }
            }
        }

        sc.close();
    }
}
